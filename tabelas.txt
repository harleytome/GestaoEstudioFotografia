tabela DIM_CLIENTES :
(
  `data_cadastro` datetime DEFAULT NULL,
  `cod_cliente` integer NOT NULL,
  `nome_completo` varchar(100) DEFAULT NULL,
  `data_nascimento` date DEFAULT NULL,
  `endereco` varchar(50) DEFAULT NULL,
  `numero` varchar(5) DEFAULT NULL,
  `bairro` varchar(30) DEFAULT NULL,
  `complemento` varchar(30) DEFAULT NULL,
  `cidade` varchar(30) DEFAULT NULL,
  `cep` varchar(10) DEFAULT NULL,
  `estado` varchar(2) DEFAULT NULL,
  `email_contato` varchar(45) DEFAULT NULL,
  `telefone` varchar(15) DEFAULT NULL,
  `celular` varchar(15) DEFAULT NULL,
  `cpf` varchar(15) DEFAULT NULL,
  `rg` varchar(15) DEFAULT NULL,
  `tipo_pessoa` char(1) DEFAULT 'F',
  `cnpj` varchar(20) DEFAULT NULL,
  `inscricao` varchar(20) DEFAULT NULL,
  `nosconheceu` varchar(45) DEFAULT NULL,
  `mailing` varchar(3) DEFAULT 'SIM',
  `observacao` varchar(200) DEFAULT NULL
)

tabela DIM_TIPO_SERVICOS :
(
	`cod_tipo_servico` integer not null,
	`descricao_tipo_servico` varchar(200) defaul null
)

tabela DIM_SERVICOS:
(
	`cod_servico` integer not null,
	`cod_tipo_servico` integer not null,
	`descricao_servico varchar(200) not null,
	`valor_sugerido` float,
	`desativado` boolean defaut false
)

tabela FATO_SERVICOS:
(
	`nr_ordem_servico` integer not null,
	`cod_cliente` integer not null,
	`cod_servico` integer not null,
	`data_servico` datetime default null,
	`descricao_servico varchar(200) not null,
	`formas_pagamento` varchar(200) not null,
	`valor_sugerido` float,
	`status_ordem_servico` varchar(100) not null, # deverá ser Em Aberto ou Cancelado ou Executado
)

tabela DIM_CONTRATOS:
(
  `nr_contrato` integer not null,

  `nome_completo` varchar(100) DEFAULT NULL,
  `data_contrato` datetime not null,
  `nome_arquivo_criado` varchar(100),
  `status_contrato` varchar(10)
